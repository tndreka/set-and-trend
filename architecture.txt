    func NewDatabase(ctx context.Context, cfg *Config) (*db.Queries, *pgxpool.Pool, error) {
        // Build connection string
        dsn := fmt.Sprintf(
            "postgres://%s:%s@%s:%s/%s?sslmode=%s",
            cfg.DBUser, cfg.DBPassword, cfg.DBHost, cfg.DBPort, cfg.DBName, cfg.DBSSLMode,
        )
        
        // Configure connection pool
        poolConfig.MaxConns = 100
        poolConfig.MinConns = 10
        poolConfig.MaxConnLifetime = time.Hour
        poolConfig.MaxConnIdleTime = 30 * time.Minute
        poolConfig.HealthCheckPeriod = time.Minute
        
        // Create pool
        pool, err := pgxpool.NewWithConfig(ctx, poolConfig)
        
        // Verify connection
        if err := pool.Ping(ctx); err != nil {
            return nil, nil, fmt.Errorf("ping: %w", err)
        }

        return db.New(pool), pool, nil
    }

WHAT TO SAY:
"Next, I establish the database connection in internal/config/database.go.

I'm using pgx, which is the fastest PostgreSQL driver for Go. Instead of creating 
one connection, I create a connection pool with:
- 100 max connections - handles concurrent requests
- 10 minimum connections - always kept open, no cold start
- 1 hour max lifetime - prevents stale connections
- Health checks every minute - detects dead connections

The function returns two things:
1. 'queries' - a type-safe SQL interface generated by SQLC
2. 'pool' - the raw connection pool for transactions

The 'defer pool.Close()' ensures connections are cleaned up when the app exits."

        // 1. Load account

        err = ValidateTradeGeometry(input.PlannedEntry, input.PlannedSL, input.PlannedTP, direction)

        if input.PlannedRiskPct > account.MaxRiskPerTradePct {
            return nil, fmt.Errorf("planned risk exceeds account max")

        // 5. Compute risk math
        riskAmount, _ := ComputeRiskAmount(balance, input.PlannedRiskPct)
        stopDistancePips, _ := ComputeStopDistancePips(stopDistance, constants.PipValueEURUSD)

    type TradeHandler struct {
        tradeService *services.TradeService
    }

    type CreateTradeRequest struct {
        AccountID      string  `json:"account_id" binding:"required,uuid"`
        CandleID       string  `json:"candle_id" binding:"required,uuid"`
        Direction      string  `json:"direction" binding:"required,oneof=LONG SHORT long short"`
        PlannedEntry   float64 `json:"planned_entry" binding:"required,gt=0"`
        PlannedSL      float64 `json:"planned_sl" binding:"required,gt=0"`
        PlannedTP      float64 `json:"planned_tp" binding:"required,gt=0"`
        PlannedRiskPct float64 `json:"planned_risk_pct" binding:"required,gt=0,lte=100"`
    }

    func (h *TradeHandler) CreateTrade(c *gin.Context) {
        var req CreateTradeRequest
        if err := c.ShouldBindJSON(&req); err != nil {
            c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
            return
        }

        accountID, _ := uuid.Parse(req.AccountID)
        candleID, _ := uuid.Parse(req.CandleID)

        trade, err := h.tradeService.CreateTrade(c.Request.Context(), services.CreateTradeInput{
            AccountID:      accountID,
            CandleID:       candleID,
            Direction:      req.Direction,
            PlannedEntry:   req.PlannedEntry,
            PlannedSL:      req.PlannedSL,
            PlannedTP:      req.PlannedTP,
            PlannedRiskPct: req.PlannedRiskPct,
        })

        if err != nil {
            c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
            return
        }

        c.JSON(http.StatusCreated, gin.H{"status": "success", "data": trade})
    }

WHAT TO SAY:
"Handlers live in internal/handlers/ and handle HTTP concerns only.

In trade_handler.go:

1. Request struct with validation tags - 'binding:required,uuid' means the field is 
   required and must be a valid UUID. Gin validates automatically.

2. ShouldBindJSON(&req) parses the JSON body and validates all fields. If validation 
   fails, I return 400 Bad Request.

3. Handler calls service with a clean input struct - the handler knows nothing about 
   business rules.

4. Return JSON response - handlers decide HTTP status codes (201 Created, 400 Bad Request).

Handlers have zero business logic. They just translate HTTP -> service calls -> HTTP responses."
